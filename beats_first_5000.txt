        // 🎵 ANALIZAR AUDIO (BPM, duración, energía)
        async function analyzeAudio(audioFile) {
            return new Promise((resolve) => {
                const audioContext = new (window.AudioContext || window.webkitAudioContext)();
                const reader = new FileReader();

                reader.onload = async (e) => {
                    try {
                        const audioBuffer = await audioContext.decodeAudioData(e.target.result);
                        const channelData = audioBuffer.getChannelData(0);
                        const duration = Math.floor(audioBuffer.duration);

                        let bpm = 0;
                        let key = null;

                        // 1. Essentia Analysis (Key & BPM)
                        if (window.Essentia && window.EssentiaWASM) {
                            try {
                                // 🔥 YIELD TO UI (Prevent freezing)
                                await new Promise(r => setTimeout(r, 50));

                                // Load WASM (It's a global factory function)
                                const wasmModule = await window.EssentiaWASM();
                                const essentia = new window.Essentia(wasmModule);

                                // Convert to Vector (Use optimized slice if song is huge? KeyExtractor needs ~60s+)
                                // 🔥 OPTIMIZE: Use first 60 seconds (sufficient for Key/BPM)
                                const maxSamples = 44100 * 60; // 60 seconds
                                const processingData = channelData.length > maxSamples ? channelData.slice(0, maxSamples) : channelData;

                                // 🔥 YIELD AGAIN
                                await new Promise(r => setTimeout(r, 10));
                                const vector = essentia.arrayToVector(processingData);

                                // A. KEY DETECTION
                                // KeyExtractor(audioVector, profileType='bgate', polyphony=true) - using defaults
                                const keyData = essentia.KeyExtractor(vector);
                                if (keyData && keyData.key && keyData.scale) {
                                    // Normalize: "C" + "major" -> "C Major"
                                    const scaleMap = { 'major': 'Major', 'minor': 'Minor' };
                                    key = `${keyData.key} ${scaleMap[keyData.scale] || keyData.scale}`;
                                }

                                // B. BPM DETECTION
                                // RhythmExtractor2013 is robust
                                const rhythm = essentia.RhythmExtractor2013(vector);
                                if (rhythm && rhythm.bpm > 0) {
                                    bpm = Math.round(rhythm.bpm);
                                }

                                // Cleanup to free WASM memory? 
                                // Essentia.js usage guides suggest explicit deletion if creating many vectors
                                // vector.delete(); // Check API if necessary. usually vector is C++ object.
                                // But JS wrapper 'arrayToVector' might handle lifecycle? 
                                // Safety: Let GC handle it or Essentia helper.
                                if (essentia.delete) essentia.delete(vector); // Hypothetical

                            } catch (err) {
                                console.warn("Essentia analysis error:", err);
                            }
                        }

                        // 2. Fallback BPM (if Essentia missing/failed)
                        if (!bpm) {
                            const peaks = detectPeaks(channelData, audioBuffer.sampleRate);
                            bpm = Math.round((peaks.length / duration) * 60);
                        }

                        // 3. Energy / Mood
                        let sum = 0;
                        // Use first 30s for energy to be fast
                        const energySlice = channelData.slice(0, 44100 * 30);
                        for (let i = 0; i < energySlice.length; i++) sum += Math.abs(energySlice[i]);
                        const avgEnergy = sum / energySlice.length;

                        let mood = 'chill';
                        if (avgEnergy > 0.15) mood = 'hard';
                        else if (avgEnergy > 0.08) mood = 'melodic';
                        else if (avgEnergy > 0.04) mood = 'ambient';

                        resolve({
                            duration,
                            bpm: Math.max(60, Math.min(250, bpm)), // Limit 250
                            key: key, // Can be null
                            energy: avgEnergy,
                            mood
                        });
                    } catch (error) {
                        console.error('Audio analysis failed:', error);
                        resolve({ duration: 0, bpm: 0, key: null, energy: 0, mood: 'unknown' });
                    }
                };

                reader.readAsArrayBuffer(audioFile);
            });
        }

        // Detectar picos de amplitud (para BPM)
        function detectPeaks(data, sampleRate) {
            const peaks = [];
            const threshold = 0.15;
            const minDistance = Math.floor(sampleRate * 0.3); // Mínimo 300ms entre picos

            for (let i = minDistance; i < data.length - minDistance; i++) {
                if (Math.abs(data[i]) > threshold) {
                    let isPeak = true;
                    for (let j = i - minDistance; j < i + minDistance; j++) {
                        if (j !== i && Math.abs(data[j]) > Math.abs(data[i])) {
                            isPeak = false;
                            break;
                        }
                    }
                    if (isPeak) {
                        peaks.push(i);
                        i += minDistance; // Saltar para evitar duplicados
                    }
                }
            }
            return peaks;
        }

        async function loadUserProfile() {

        function validateReleaseDate() {
            const dateInput = document.getElementById('dateInput');
            const dateError = document.getElementById('dateError');

            if (!dateInput.value) return false;

            // ðŸ”¥ FIX: Parse date parts manually to ensure local time midnight
            // "2025-12-18" -> [2025, 12, 18]
            const parts = dateInput.value.split('-');
            const selected = new Date(parts[0], parts[1] - 1, parts[2]); // Year, Month(0-11), Day

            // Normalize "now" to midnight today
            const now = new Date();
            now.setHours(0, 0, 0, 0);

            // ðŸ”¥ PROGRAMACIÃ“N DE FECHA: Validar SOLO si el usuario intenta cambiar la fecha
            // Si ya existe una fecha vÃ¡lida y no la estÃ¡ cambiando, NO validar contra el presente
            const isEditingExistingDate = dateInput.dataset.originalDate && dateInput.value === dateInput.dataset.originalDate;

            if (isEditingExistingDate) {
                // âœ… La fecha ya fue guardada antes - NO validar contra presente
                if (dateError) dateError.style.display = 'none';
                dateInput.classList.remove('error');
                return true;
            }

            // âœ… Comparar timestamps (ambos estÃ¡n a las 00:00:00)
            if (selected.getTime() < now.getTime()) {
                if (dateError) {
                    dateError.innerText = "La fecha de lanzamiento no puede ser en el pasado";
                    dateError.style.display = 'block';
                }
                dateInput.classList.add('error');
                return false;
            } else {
                if (dateError) dateError.style.display = 'none';
                dateInput.classList.remove('error');
                return true;
            }
        }
