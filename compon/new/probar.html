Error State Shake

Transición de error y sacudida horizontal multi-fase calculada con curvas cubic-bezier independientes. Se auto-restaura automáticamente o se limpia al tipear.

Vista Previa Interactiva Haz clic en "Animar Error" o presiona Enter

Por favor ingresa un correo válido.

/**
 * OFFSZN Helper: Trigger Error State Shake
 * @param {HTMLElement} wrapEl - El contenedor .t-input-wrap
 * @param {string} [customMessage] - Mensaje de error opcional
 */
function triggerInputShake(wrapEl, customMessage) {
  if (!wrapEl) return;
  const input = wrapEl.querySelector('.t-input');
  const msgEl = wrapEl.querySelector('.t-error-msg');
  if (customMessage && msgEl) msgEl.textContent = customMessage;

  // Activar estado de error
  wrapEl.classList.add('is-error');
  input.classList.add('is-error');

  // Reiniciar animación de sacudida mediante reflow
  input.classList.remove('is-shaking');
  void input.offsetWidth; // Force reflow
  input.classList.add('is-shaking');

  // Duración total del shake (80ms*2 + 60ms*2 = 280ms)
  const shakeMs = 280;
  window.setTimeout(() => {
    input.classList.remove('is-shaking');
  }, shakeMs + 20);

  // Auto-revert si no interactúa el usuario
  if (wrapEl._revertTimer) window.clearTimeout(wrapEl._revertTimer);
  const holdMs = 3000;
  wrapEl._revertTimer = window.setTimeout(() => {
    wrapEl._revertTimer = null;
    wrapEl.classList.remove('is-error');
    input.classList.remove('is-error');
  }, shakeMs + holdMs);
}

// Inicializar auto-limpieza al tipear
document.querySelectorAll('.t-input-wrap').forEach((wrap) => {
  const inputEl = wrap.querySelector('.t-input-el');
  const input = wrap.querySelector('.t-input');
  if (!inputEl) return;

  inputEl.addEventListener('input', () => {
    if (wrap._revertTimer) {
      window.clearTimeout(wrap._revertTimer);
      wrap._revertTimer = null;
    }
    wrap.classList.remove('is-error');
    input.classList.remove('is-error');
  });
});
import { useRef, useState } from "react";

export function InputShake({ children, message = "Campo inválido", onCancel }) {
  const inputRef = useRef(null);
  const wrapRef = useRef(null);
  const timerRef = useRef(null);
  const [hasError, setHasError] = useState(false);

  const triggerError = () => {
    if (!inputRef.current) return;
    setHasError(true);
    inputRef.current.classList.remove("is-shaking");
    void inputRef.current.offsetWidth;
    inputRef.current.classList.add("is-shaking");

    window.setTimeout(() => {
      inputRef.current?.classList.remove("is-shaking");
    }, 300);

    if (timerRef.current) clearTimeout(timerRef.current);
    timerRef.current = setTimeout(() => {
      setHasError(false);
    }, 3300);
  };

  const handleInput = () => {
    if (timerRef.current) clearTimeout(timerRef.current);
    setHasError(false);
    onCancel?.();
  };

  return (
    <div ref={wrapRef} className={`t-input-wrap ${hasError ? "is-error" : ""}`}>
      <div ref={inputRef} className={`t-input ${hasError ? "is-error" : ""}`} onInput={handleInput}>
        {children}
      </div>
      <p className="t-error-msg">{message}</p>
    </div>
  );
}
